### 3.2 property——最常用的数据描述符

`property`是Python内置的数据描述符实现，是"Pythonic"封装的标准方式。

**property的本质**：

```python
# property 是一个C实现的类，等价的纯Python实现：

class Property:
    "模拟内置 property 的纯Python实现"
    def __init__(self, fget=None, fset=None, fdel=None, doc=None):
        self.fget = fget
        self.fset = fset
        self.fdel = fdel
        self.__doc__ = doc or (fget.__doc__ if fget else None)

    def __get__(self, obj, objtype=None):
        if obj is None:
            return self  # 类级别访问返回property本身
        if self.fget is None:
            raise AttributeError("unreadable attribute")
        return self.fget(obj)

    def __set__(self, obj, value):
        if self.fset is None:
            raise AttributeError("can't set attribute")
        self.fset(obj, value)

    def __delete__(self, obj):
        if self.fdel is None:
            raise AttributeError("can't delete attribute")
        self.fdel(obj)

    def getter(self, fget):
        return type(self)(fget, self.fset, self.fdel, self.__doc__)

    def setter(self, fset):
        return type(self)(self.fget, fset, self.fdel, self.__doc__)

    def deleter(self, fdel):
        return type(self)(self.fget, self.fset, fdel, self.__doc__)
```

**使用场景**：

```python
class Circle:
    def __init__(self, radius):
        self.radius = radius  # 注意：这会触发 property.__set__！

    @property
    def area(self):
        """计算属性：每次访问时计算"""
        import math
        return math.pi * self.radius ** 2

    @property
    def radius(self):
        """带验证的属性"""
        return self._radius

    @radius.setter
    def radius(self, value):
        if value < 0:
            raise ValueError("半径不能为负数")
        self._radius = value

    @radius.deleter
    def radius(self):
        print("半径被删除，重置为0")
        self._radius = 0

c = Circle(5)
print(c.area)        # 78.54... — 计算属性
c.radius = -1        # ValueError!
del c.radius         # 半径被删除，重置为0

# 为什么要用property而不是直接用属性？
# 1. 封装：内部实现可以改变，但API不变
# 2. 验证：设置值时可以做校验
# 3. 计算：惰性计算属性
# 4. 只读：不定义setter就是只读属性
# 5. 日志/通知：在get/set中添加副作用
```

**只读property的陷阱**：property只读是指**不能通过property协议赋值**，但如果实例有`__dict__`，仍然可以直接改内部变量：

```python
class ReadOnly:
    @property
    def value(self):
        return self._value

    def __init__(self, v):
        self._value = v

ro = ReadOnly(42)
ro.value = 99       # AttributeError: can't set attribute ✓
ro._value = 99       # 但这可以！_value不是property只是普通实例属性
print(ro.value)      # 99

# 真正的只读需要用 __slots__ 或其他机制
```

**实验：用ctypes窥探property在CPython中的slot结构**

```python
import ctypes

# 在CPython中，property 是 PyGetSetDef 的Python包装
# 它是一个C结构体，包含 getter/setter 函数指针

class MyProp:
    @property
    def x(self):
        return 42

# 检查 property 的类型槽
prop_type = type(MyProp.x)  # <class 'property'>

# property 定义了 __get__ 和 __set__（即使没有setter，__set__也存在）
# 所以它是数据描述符
print(hasattr(prop_type, '__get__'))   # True
print(hasattr(prop_type, '__set__'))   # True（这使它成为数据描述符！）

# 但是，只读property的 __set__ 在调用时会抛出 AttributeError
# CPython内部：property.__set__ 检查 fset 是否为 NULL
try:
    MyProp().x = 10  # 调用 property.__set__，但fset=None → AttributeError
except AttributeError as e:
    print(f"Expected error: {e}")
```

### 3.3 classmethod与staticmethod

`classmethod`和`staticmethod`都是**非数据描述符**，通过`__get__`改变绑定的行为。

```python
class Demo:
    @classmethod
    def cls_method(cls, x):
        return f"classmethod: cls={cls.__name__}, x={x}"

    @staticmethod
    def static_method(x):
        return f"staticmethod: x={x}"

    def instance_method(self, x):
        return f"instance method: self={type(self).__name__}, x={x}"

d = Demo()

# classmethod: __get__ 返回绑定到类的方法
print(Demo.cls_method(42))    # classmethod: cls=Demo, x=42
print(d.cls_method(42))       # classmethod: cls=Demo, x=42（即使通过实例调用，cls也是Demo）

# staticmethod: __get__ 返回原始函数，无绑定
print(Demo.static_method(42)) # staticmethod: x=42
print(d.static_method(42))    # staticmethod: x=42

# 检查类型
print(type(Demo.cls_method))   # <class 'builtin_function_or_method'>
print(type(Demo.static_method))  # <class 'function'>
```

**纯Python等价实现**：

```python
class ClassMethod:
    "模拟 classmethod"
    def __init__(self, func):
        self.func = func
        self.__doc__ = func.__doc__

    def __get__(self, obj, objtype=None):
        if objtype is None:
            objtype = type(obj)
        # 返回绑定到类的方法（不是实例！）
        # classmethod 将 cls 作为第一个参数
        import functools
        return functools.partial(self.func, objtype)

class StaticMethod:
    "模拟 staticmethod"
    def __init__(self, func):
        self.func = func
        self.__doc__ = func.__doc__

    def __get__(self, obj, objtype=None):
        # 返回原始函数，不做任何绑定
        return self.func

# 验证
class TestCM:
    @ClassMethod
    def hello(cls, name):
        return f"Hello from {cls.__name__}, {name}!"

class TestSM:
    @StaticMethod
    def hello(name):
        return f"Hello, {name}!"

print(TestCM.hello("World"))  # Hello from TestCM, World!
print(TestSM.hello("World"))  # Hello, World!
```

**关键理解**：`classmethod`和`staticmethod`都是通过`__get__`描述符协议实现不同绑定行为——classmethod绑定到类，staticmethod不做绑定。两者都是**非数据描述符**，所以实例`__dict__`可以覆盖它们。

**实验：覆盖classmethod和staticmethod**

```python
class Override:
    @classmethod
    def cm(cls):
        return "classmethod"

    @staticmethod
    def sm():
        return "staticmethod"

o = Override()

# 实例可以覆盖（因为是非数据描述符）
o.cm = lambda: "instance cm"
o.sm = lambda: "instance sm"

print(o.cm())        # instance cm — 实例__dict__覆盖了classmethod
print(o.sm())        # instance sm — 实例__dict__覆盖了staticmethod
print(Override.cm()) # classmethod — 类级别仍正常
print(Override.sm()) # staticmethod — 类级别仍正常
```

### 3.4 __slots__——内存优化的代价

`__slots__`是Python中显著的内存优化手段，其原理是用**固定偏移量的C数组**替代`__dict__`。

```python
# 不使用 __slots__
class NoSlot:
    def __init__(self, x, y):
        self.x = x
        self.y = y

# 使用 __slots__
class WithSlot:
    __slots__ = ('x', 'y')
    def __init__(self, x, y):
        self.x = x
        self.y = y

# 内存对比
import sys

ns = NoSlot(1, 2)
ws = WithSlot(1, 2)

print(f"NoSlot 实例: {sys.getsizeof(ns)} 字节")  # ~264字节（包含__dict__引用）
print(f"  __dict__: {sys.getsizeof(ns.__dict__)} 字节")  # ~104字节（dict本身）
print(f"WithSlot 实例: {sys.getsizeof(ws)} 字节")  # ~56字节！
# 总节约: 约 264 + 104 - 56 = 312 字节/实例

# 创建100万个实例的内存差异
# NoSlot: ~368 MB
# WithSlot: ~56 MB
# 差异：约6.5倍！
```

**CPython中__slots__的实现**：

```c
// __slots__ 的核心原理：
// 1. 类定义 __slots__ 时，tp_dictoffset 被设置为 0（或-1），表示没有 __dict__
// 2. 每个 slot 成为一个 member descriptor（PyMemberDef）
// 3. member descriptor 是数据描述符，存储在类的 __dict__ 中
// 4. 实例的属性值存储在固定偏移量位置，而不是 __dict__ 中

// PyMemberDef 结构
typedef struct {
    const char *name;           // 属性名 "x"
    int type;                   // 类型 T_INT, T_OBJECT, ...
    Py_spath_t offset;          // 在实例中的偏移量
    int flags;                  // READONLY 等
    const char *doc;            // 文档字符串
} PyMemberDef;

// 例如 WithSlot 的内存布局:
// PyObject_HEAD (16字节: refcnt + type pointer)
// slot 'x' at offset 16  (一个 PyObject*，8字节)
// slot 'y' at offset 24  (一个 PyObject*，8字节)
// 总共: 16 + 8 + 8 = 32，加上对齐可能48-56字节
```

**实验：__slots__的偏移量和描述符机制**

```python
class Point3D:
    __slots__ = ('x', 'y', 'z')

# 每个 slot 是一个 member_descriptor（数据描述符）
for name in Point3D.__slots__:
    desc = getattr(Point3D, name)
    print(f"{name}: type={type(desc).__name__}, offset={desc.offset}")
# x: type=member_descriptor, offset=16
# y: type=member_descriptor, offset=24
# z: type=member_descriptor, offset=32

# member_descriptor 同时定义了 __get__ 和 __set__
# → 它是数据描述符 → 优先于实例 __dict__
# 但 __slots__ 类没有 __dict__，所以这个优先级不太会体现

p = Point3D(1, 2, 3)
# p.x → Point3D.x.__get__(p, Point3D)
#       → 从 p 的偏移量 16 处读取 PyObject*

# 尝试添加非slot属性
try:
    p.w = 4  # AttributeError: 'Point3D' object has no attribute 'w'
except AttributeError as e:
    print(f"Expected: {e}")

# 没有实例 __dict__
print(hasattr(p, '__dict__'))  # False
```

**__slots__的继承陷阱**：

```python
class Base:
    __slots__ = ('x',)

class Child(Base):
    # 继承了Base的slots，但Child有自己的 __dict__
    # 除非Child也定义 __slots__
    pass

c = Child()
c.x = 1     # OK，通过Base的slot
c.y = 2     # OK！Child有__dict__（因为没定义__slots__）
print(c.__dict__)  # {'y': 2}

# 要使子类也没有 __dict__，子类也必须定义 __slots__
class ChildNoDict(Base):
    __slots__ = ('y',)  # 只需列出新增的slot

c2 = ChildNoDict()
c2.x = 1     # 通过Base的slot
c2.y = 2     # 通过自己的slot
try:
    c2.z = 3
except AttributeError:
    print("No __dict__!")

# __slots__ 为空元组 = 只有继承的slot，无新增slot，也无 __dict__
class ChildEmptySlot(Base):
    __slots__ = ()

ce = ChildEmptySlot()
ce.x = 1     # 通过Base的slot
try:
    ce.y = 2
except AttributeError:
    print("No __dict__!")
```

**__slots__与__dict__共存**：

```python
class Hybrid:
    __slots__ = ('x', 'y', '__dict__')  # __dict__ 也可以作为slot
    # 现在有slots x, y，也有动态 __dict__

h = Hybrid()
h.x = 1         # 通过slot
h.y = 2         # 通过slot
h.z = 3         # 通过__dict__！
print(h.__dict__)  # {'z': 3}
# 这样可以兼得slot的内存优化和__dict__的灵活性
```

### 3.5 函数也是描述符

这是Python OOP最精妙的机制之一：**方法的绑定行为完全由函数的描述符协议实现**。

```python
class Foo:
    def method(self):
        return "hello"

# Foo.method 本质上是一个函数对象
# 函数类型定义了 __get__ → 函数是非数据描述符

f = Foo()

# f.method 的过程：
# 1. Python 查找 f 的属性 'method'
# 2. 在 Foo.__dict__ 中找到 'method' → 是一个函数对象
# 3. 函数是非数据描述符（只有__get__，没有__set__）
# 4. 先检查 f.__dict__ → 没有 'method'
# 5. 调用 Foo.__dict__['method'].__get__(f, Foo)
#    → 返回 bound method object

# 类级别访问：Foo.method
# 调用 Foo.__dict__['method'].__get__(None, Foo)
# 当 obj=None 时，返回函数本身
print(Foo.method)          # <function Foo.method at 0x...>
print(type(Foo.method))     # <class 'function'>

# 实例级别访问：f.method
print(type(f.method))       # <class 'method'>
```

**手动模拟方法绑定——不使用class语法**：

```python
def greet(self, name):
    return f"Hello, {name}! I'm {self.name}"

# 手动创建类（不用class语句）
Person = type('Person', (), {
    'name': 'World',
    'greet': greet,
})

p = Person()

# Person.greet 是一个函数（非数据描述符）
print(type(Person.__dict__['greet']))  # <class 'function'>

# p.greet 是一个绑定方法
print(type(p.greet))  # <class 'method'>

# 手动触发描述符协议
raw_func = Person.__dict__['greet']
bound = raw_func.__get__(p, Person)  # 手动绑定
print(bound("Alice"))  # Hello, Alice! I'm World

# 未绑定时直接调用
print(raw_func(p, "Bob"))  # Hello, Bob! I'm World
```

**函数的`__get__`在CPython中的实现**：

```c
// CPython Objects/funcobject.c
static PyObject *
func_descr_get(PyObject *func, PyObject *obj, PyObject *type)
{
    if (obj == Py_None || obj == NULL) {
        // obj 为 None 或 NULL → 类级别访问 → 返回函数本身
        Py_INCREF(func);
        return func;
    }
    // obj 是实例 → 返回绑定方法
    return PyMethod_New(func, obj);
}

// 对应Python中的:
// function.__get__(None, type) → function itself
// function.__get__(obj, type) → bound method(func, obj)
```

Python的魔术方法（dunder methods）不是语法糖——它们是与C层类型槽（type slots）的直接映射。

### 4.2 运算符到slot调用的映射过程

```python
# 当你写 a + b 时，Python做了什么？

# 字节码:
import dis
dis.dis(compile("a + b", '<test>', 'eval'))
# 1           0 LOAD_NAME                0 (a)
#             2 LOAD_NAME                1 (b)
#             4 BINARY_ADD

# BINARY_ADD 的执行逻辑（简化版）:
# 1. 检查 type(a) 是否有 nb_add slot
# 2. 如果有，调用 type(a).__add__(a, b)
# 3. 如果返回 NotImplemented，尝试 type(b).__radd__(b, a)
# 4. 如果仍然 NotImplemented → TypeError
```

**CPython中BINARY_ADD的实现**：

```c
// CPython Python/ceval.c （简化版）
case TARGET(BINARY_ADD): {
    PyObject *right = POP();
    PyObject *left = TOP();
    PyObject *sum;

    // 快速路径：两者都是int
    if (PyLong_CheckExact(left) && PyLong_CheckExact(right)) {
        sum = long_add((PyLongObject *)left, (PyLongObject *)right);
    }
    // 快速路径：两者都是str (拼接)
    else if (PyUnicode_CheckExact(left) && PyUnicode_CheckExact(right)) {
        sum = unicode_concat(left, right);
    }
    // 通用路径：通过type slot调用
    else {
        sum = PyNumber_Add(left, right);
    }

    if (sum == NULL) goto error;
    SET_TOP(sum);
    DISPATCH();
}

// PyNumber_Add 的逻辑：
PyObject *PyNumber_Add(PyObject *v, PyObject *w)
{
    PyNumberMethods *v_methods = Py_TYPE(v)->tp_as_number;
    // 1. 尝试 v.__add__(w)
    if (v_methods && v_methods->nb_add) {
        PyObject *result = (*v_methods->nb_add)(v, w);
        if (result != Py_NotImplemented) return result;
        Py_DECREF(result);
    }
    // 2. 尝试 w.__radd__(v)（reflected/add method）
    // ...
    // 3. TypeError
}
```

### 4.3 实验：自定义Vector类的运算符

```python
import math

class Vector:
    """2D向量，演示魔术方法与C slot映射"""

    __slots__ = ('_x', '_y')

    def __init__(self, x=0.0, y=0.0):
        self._x = float(x)
        self._y = float(y)

    @property
    def x(self):
        return self._x

    @property
    def y(self):
        return self._y

    # ---- 对象表示 (tp_repr, tp_str) ----
    def __repr__(self):
        return f"Vector({self._x}, {self._y})"

    def __str__(self):
        return f"({self._x:.2f}, {self._y:.2f})"

    # ---- 比较运算 (tp_richcompare) ----
    def __eq__(self, other):
        if not isinstance(other, Vector):
            return NotImplemented
        return self._x == other._x and self._y == other._y

    def __lt__(self, other):
        if not isinstance(other, Vector):
            return NotImplemented
        return self.magnitude() < other.magnitude()

    def __le__(self, other):
        if not isinstance(other, Vector):
            return NotImplemented
        return self.magnitude() <= other.magnitude()

    # ---- 哈希 (tp_hash) ----
    def __hash__(self):
        return hash((self._x, self._y))

    # ---- 布尔值 (nb_bool) ----
    def __bool__(self):
        return self._x != 0 or self._y != 0

    # ---- 算术运算 (nb_add, nb_sub, nb_mul, nb_negative) ----
    def __add__(self, other):
        if isinstance(other, Vector):
            return Vector(self._x + other._x, self._y + other._y)
        return NotImplemented

    def __radd__(self, other):
        # 支持 0 + vector 的情况
        return self.__add__(other)

    def __sub__(self, other):
        if isinstance(other, Vector):
            return Vector(self._x - other._x, self._y - other._y)
        return NotImplemented

    def __mul__(self, other):
        # 标量乘法
        if isinstance(other, (int, float)):
            return Vector(self._x * other, self._y * other)
        # 点积
        if isinstance(other, Vector):
            return self._x * other._x + self._y * other._y
        return NotImplemented

    def __rmul__(self, other):
        # 支持 3 * vector
        if isinstance(other, (int, float)):
            return Vector(self._x * other, self._y * other)
        return NotImplemented

    def __neg__(self):
        return Vector(-self._x, -self._y)

    def __abs__(self):
        return self.magnitude()

    # ---- 长度 (mp_length) ----
    def __len__(self):
        return 2

    # ---- 索引 (sq_item / mp_subscript) ----
    def __getitem__(self, index):
        if index == 0: return self._x
        if index == 1: return self._y
        raise IndexError(f"Vector index out of range: {index}")

    # ---- 迭代 (tp_iter) ----
    def __iter__(self):
        yield self._x
        yield self._y

    # ---- 辅助方法 ----
    def magnitude(self):
        return math.sqrt(self._x ** 2 + self._y ** 2)

# 测试
v1 = Vector(3, 4)
v2 = Vector(1, 2)

print(repr(v1))          # Vector(3.0, 4.0)    → tp_repr
print(str(v1))           # (3.00, 4.00)        → tp_str
print(v1 + v2)           # Vector(4.0, 6.0)    → nb_add
print(v1 * 2)            # Vector(6.0, 8.0)    → nb_multiply
print(3 * v1)            # Vector(9.0, 12.0)   → nb_multiply (via __rmul__)
print(v1 * v2)           # 11.0                → nb_multiply (dot product)
print(-v1)               # Vector(-3.0, -4.0)  → nb_negative
print(abs(v1))           # 5.0                 → nb_absolute
print(bool(v1))          # True                → nb_bool
print(bool(Vector(0,0))) # False               → nb_bool
print(v1 == v2)          # False               → tp_richcompare (Py_EQ)
print(v1 < v2)           # False               → tp_richcompare (Py_LT)
print(len(v1))           # 2                   → mp_length
print(v1[0], v1[1])      # 3.0 4.0            → mp_subscript
x, y = v1               # 3.0, 4.0            → tp_iter
print(hash(v1))          # 某个int             → tp_hash
```

### 4.4 魔术方法的"子protocol"链

许多魔术方法形成**协议链**——如果一个没定义，Python会回退到另一个：

```python
# bool() 的回退链：
# 1. type(x).__bool__(x)  → nb_bool
# 2. type(x).__len__(x) > 0  → mp_length (如果__bool__未定义)
# 3. 默认 True

# iter() 的回退链：
# 1. type(x).__iter__(x)  → tp_iter
# 2. type(x).__getitem__(x)  → 从0开始递增调用，直到IndexError

# 反向运算的回退链（a + b）：
# 1. type(a).__add__(a, b)  → nb_add
# 2. 若返回 NotImplemented → type(b).__radd__(b, a)

# 增量运算的映射（a += b）：
# 1. type(a).__iadd__(a, b)  → nb_inplace_add
# 2. 若未定义 → type(a).__add__(a + b) 的结果赋回a

class NoIter:
    """演示 iter() 的 __getitem__ 回退"""
    def __init__(self, data):
        self._data = data

    def __getitem__(self, index):
        if index >= len(self._data):
            raise IndexError
        return self._data[index]

ni = NoIter([10, 20, 30])
for x in ni:
    print(x)
# 10
# 20
# 30
# iter() 回退到 __getitem__，从index=0开始递增调用
```

## 五、常见陷阱

### 陷阱1：可变类属性被所有实例共享（直到实例创建了自己的副本）

```python
class Trap1:
    items = []  # 可变类属性——危险！

a = Trap1()
b = Trap1()

a.items.append('a')
print(b.items)  # ['a']  — b看到了a的修改！因为a.items查找到的是类的items

# 为什么？属性查找过程：
# a.items → a.__dict__ 没有 'items' → Trap1.__dict__ 有 'items' → 返回类属性的列表
# a.items.append() → 修改的是类属性列表，不是实例属性

# 当实例"覆盖"时：
a.items = ['new']  # 这会在 a.__dict__ 中创建 'items'
print(a.items)      # ['new']  — 现在是实例属性
print(b.items)      # ['a']    — 仍指向类属性
print(Trap1.items)  # ['a']    — 类属性没变

# 解决方案：在 __init__ 中创建实例属性
class Fixed1:
    def __init__(self):
        self.items = []  # 每个实例独立的列表

a = Fixed1()
b = Fixed1()
a.items.append('a')
print(b.items)  # []  — b有自己的列表
```

**根因**：可变类属性是**共享状态**，实例属性查找规则（类属性在实例`__dict__`中找不到时回退到类`__dict__`）使得多个实例看到同一个可变对象。

### 陷阱2：super()在__init__中不使用协作模式

```python
class Base:
    def __init__(self):
        print("Base.__init__")

class A(Base):
    def __init__(self):
        Base.__init__(self)  # 直接调用！不是super()
        print("A.__init__")

class B(Base):
    def __init__(self):
        Base.__init__(self)  # 直接调用！不是super()
        print("B.__init__")

class C(A, B):
    def __init__(self):
        A.__init__(self)  # 会调用 Base.__init__
        B.__init__(self)  # 又调用 Base.__init__
        print("C.__init__")

c = C()
# Base.__init__  ← 被调用了两次！
# A.__init__
# Base.__init__  ← 重复调用
# B.__init__
# C.__init__

# 修正：使用 super() 的协作模式
class Base2:
    def __init__(self, **kwargs):
        super().__init__(**kwargs)  # 始终调用super
        print("Base2.__init__")

class A2(Base2):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        print("A2.__init__")

class B2(Base2):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        print("B2.__init__")

class C2(A2, B2):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        print("C2.__init__")

c2 = C2()
# Base2.__init__
# B2.__init__
# A2.__init__
# C2.__init__
# Base2.__init__ 只被调用一次！MRO保证了线性化。
```

**根因**：直接调用`ClassName.__init__(self)`绕过了MRO链，每个类独立指向`Base`， 导致`Base.__init__`被多次调用。`super()`遵循MRO线性链，保证每个`__init__`恰好调用一次。

### 陷阱3：__slots__与继承的微妙交互

```python
class A:
    __slots__ = ('x',)

class B(A):
    __slots__ = ('y',)

class C(A):
    __slots__ = ('__dict__',)  # C重新引入了 __dict__

class D(A):
    pass  # 忘记定义 __slots__ → 自动有 __dict__

b = B()
b.x = 1  # OK — 通过A的slot
b.y = 2  # OK — 通过B的slot
try:
    b.z = 3
except AttributeError:
    print("b没有__dict__，不能添加z")  # 正确行为

c = C()
c.x = 1     # OK — 通过A的slot
c.anything = 42  # OK！C有 __dict__ slot

d = D()
d.x = 1     # OK — 通过A的slot
d.anything = 42  # OK！D有自动创建的 __dict__

# 更危险的变体：混合继承
class E(B, C):
    __slots__ = ('z',)

# MRO: E → B → C → A → object
# B没有__dict__，C有__dict__
# E.__dictoffset 从C继承 → E有__dict__！
e = E()
e.x = 1
e.y = 2
e.z = 3
e.dynamic = 99  # 可以！因为C引入了__dict__
```

**根因**：`__slots__`的效果不自动传递给子类——子类必须**显式声明`__slots__`**才能禁用`__dict__`。如果忘记声明，子类会自动拥有`__dict__`，失去内存优化效果。混合继承时，任何一个基类有`__dict__`，子类也会有。

### 陷阱4：描述符放在实例级别不起作用

```python
class MyDescriptor:
    def __get__(self, obj, objtype=None):
        return "descriptor value"
    def __set__(self, obj, value):
        print(f"descriptor set: {value}")

class Trap4:
    pass

# 错误：将描述符放在实例上
t = Trap4()
t.desc = MyDescriptor()  # 描述符被存在实例 __dict__ 中

print(t.desc)  # <__main__.MyDescriptor object at 0x...>  ← 不是描述符行为！
# 返回的是描述符对象本身，而不是调用 __get__ 的结果

# 为什么？描述符协议只在"类型__dict__"中生效！
# 属性查找中的描述符检查只搜索 type(obj).__mro__ 中的类 __dict__
# 实例 __dict__ 中的描述符对象不会被识别为描述符

# 正确做法：将描述符放在类级别
class Fixed4:
    desc = MyDescriptor()  # 放在类__dict__中

f = Fixed4()
print(f.desc)  # descriptor value  ← 正确的描述符行为！
```

**根因**：CPython的属性查找（`_PyObject_GenericGetAttrWithDict`）只在`type(obj)`的MRO链中检查描述符协议。实例`__dict__`中的对象是"普通属性"，不经过描述符协议检查。这是设计选择——如果实例`__dict__`也要检查描述符，每次属性访问的开销会显著增加。
