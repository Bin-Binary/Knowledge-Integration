#### 替换 sys.stdout 对象

```python
import sys
import io

# 保存原始 stdout
original_stdout = sys.stdout

# 替换为自定义对象
class TeeOutput:
    """同时输出到多个目标的 Tee 类"""
    def __init__(self, *targets):
        self.targets = targets
    
    def write(self, msg):
        for target in self.targets:
            target.write(msg)
    
    def flush(self):
        for target in self.targets:
            target.flush()

log_file = open("output.log", "w")
sys.stdout = TeeOutput(original_stdout, log_file)

print("This goes to both terminal and file")

sys.stdout = original_stdout  # 恢复
log_file.close()
```

### 5.1 os.path：函数式路径操作（传统）

```python
import os
```

### 5.2 pathlib.Path：面向对象的现代方式

pathlib 在 Python 3.4 引入，提供了更优雅、更安全的路径操作方式。

```python
from pathlib import Path

# 创建 Path 对象
p = Path("/home/user/documents/file.txt")

# 路径组件
print(p.name)      # 'file.txt'
print(p.stem)      # 'file' (不含后缀)
print(p.suffix)    # '.txt'
print(p.suffixes)  # ['.txt'] (多层后缀如 .tar.gz → ['.tar', '.gz'])
print(p.parent)    # Path('/home/user/documents')
print(p.parents)   # 序列: parents[0]=..., parents[1]=...父路径链

# 路径拼接 (使用 / 运算符!)
new_path = p.parent / "another.txt"
print(new_path)  # /home/user/documents/another.txt

# 绝对与相对
print(p.is_absolute())      # True
print(Path("relative.txt").resolve())  # 转为绝对路径

# 文件系统操作
print(p.exists())    # True/False
print(p.is_file())   # 是否为文件
print(p.is_dir())    # 是否为目录

# 读写文件 (内置方法!)
content = p.read_text(encoding='utf-8')     # 自动打开/读取/关闭
p.write_text("new content", encoding='utf-8')  # 自动打开/写入/关闭

# 二进制
data = p.read_bytes()
p.write_bytes(data)

# 目录遍历
home = Path.home()
for f in home.iterdir():        # 遍历目录
    if f.is_file():
        print(f.name)

# 递归查找
for py_file in home.rglob("*.py"):  # 递归查找 .py 文件
    print(py_file)

for item in home.glob("**/*.py"):  # 等价写法
    print(item)

# 创建和删除
Path("new_dir").mkdir(exist_ok=True, parents=True)
Path("file.txt").unlink(missing_ok=True)  # 删除文件
Path("empty_dir").rmdir()  # 删除空目录

# 文件信息
stat = p.stat()
print(f"大小: {stat.st_size}")
```

#### os.path vs pathlib 对比

```python
import os.path

# 传统 os.path 方式
```

## 六、常见陷阱

### 陷阱 1：忘记关闭文件（资源泄漏）

```python
# ❌ 错误：文件未关闭
f = open("data.txt", "w")
f.write("important data")
# 如果这里抛异常，文件永远不会关闭
# 文件描述符泄漏，最终触及 OS 限制 (ulimit -n / _setmaxstdio)
process_data()  # 可能抛异常 → fd 泄漏
f.close()       # 永远到不了

# ✅ 正确：with 语句保证关闭
with open("data.txt", "w") as f:
    f.write("important data")
    process_data()  # 即使异常，__exit__ 也会 close()
# f 此时已关闭
```

**深层原因**：CPython 的垃圾回收**可能**最终回收未关闭的 FileIO 对象（`__del__` 会调用 `close()`），但这不可靠且时机不定。更关键的是：

```python
import os

# 在 CPython 中, 引用计数通常能及时回收, 但这不可移植
f = open("data.txt")
del f  # CPython: 立刻调用 __del__ → close()
       # PyPy/Jython: GC 时机不确定, 文件可能长时间未关闭

# 验证文件描述符泄漏
def leak_fds():
    for i in range(1020):
        f = open("/dev/null")  # 忘记关闭
    # OSError: [Errno 24] Too many open files

# Windows 上默认上限仅 512 (_setmaxstdio)
# Linux 上默认 1024 (ulimit -n)
```

CPython 的 `FileIO.__del__` 还会打印 `ResourceWarning`（在 `python -W all` 或开发模式下）：

```python
# python -W dev leak.py
# ResourceWarning: unclosed file <_io.TextIOWrapper ...>
```

### 陷阱 2：文本模式处理二进制文件（数据损坏）

```python
# ❌ 致命错误：用文本模式读取 PNG 图片
with open("photo.png", "r") as f:     # 默认文本模式!
    data = f.read()
# UnicodeDecodeError: 'utf-8' codec can't decode byte 0x89...
# 即使不报错, 数据已经过编码转换, 二进制数据被破坏

# ❌ 同样致命：用文本模式写入二进制数据
with open("output.bin", "w") as f:
    f.write(b"\x89PNG")     # TypeError: write() argument must be str, not bytes

# ❌ 更隐蔽的错误：文本模式的换行符转换损坏二进制数据
with open("data.bin", "w") as f:     # Windows 上:
    f.write("\n")                     # 写入 \r\n (2 字节!) 而非 \n (1 字节)
# 文件大小比预期多了 N 个 \r (N = 换行符数量)

# ✅ 正确：二进制模式
with open("photo.png", "rb") as f:
    data = f.read()        # bytes, 原封不动
with open("output.bin", "wb") as f:
    f.write(b"\x89PNG")   # OK
```

**规则**：只要文件不是人类可读文本（图片、视频、压缩包、pickle、数据库文件等），**必须用 `"rb"`/`"wb"`**。

### 陷阱 3：跨平台编码不一致

```python
# 同一份代码, 不同平台, 不同行为

# 写入文件
with open("data.txt", "w") as f:    # Windows: cp936, Linux: utf-8
    f.write("中文内容")

# 读取文件
with open("data.txt", "r") as f:    # 编码不一致 → UnicodeDecodeError
    content = f.read()
# Windows 写入 cp936, Linux 用 utf-8 读 → 乱码或报错

# ✅ 永远显式指定 encoding
with open("data.txt", "w", encoding="utf-8") as f:
    f.write("中文内容")
with open("data.txt", "r", encoding="utf-8") as f:
    content = f.read()   # 任何平台都一致
```

**默认编码表**（`open()` 不指定 encoding 时）：

| 平台 | `locale.getpreferredencoding()` | 常见编码 |
|:-----|:-------------------------------|:---------|
| Linux (en_US.UTF-8) | `UTF-8` | UTF-8 |
| macOS | `UTF-8` | UTF-8 |
| Windows (中文) | `cp936` | GBK |
| Windows (日文) | `cp932` | Shift-JIS |
| Windows (英文) | `cp1252` | Windows-1252 |

**Python 3.15+ 变化**：PEP 686 将 `open()` 的默认编码改为 UTF-8（可配置回退），这是对跨平台兼容性的重大改善。

### 陷阱 4：r+ 模式下未正确 seek 导致数据错位

```python
# r+ = 读写模式, 不截断文件, 写入从当前位置开始

with open("data.txt", "w") as f:
    f.write("ABCDEFGH")  # 8 字节

# ❌ 读取后直接写入, 覆盖位置不确定
with open("data.txt", "r+") as f:
    content = f.read(4)  # 读取 "ABCD", 位置移到 4
    f.write("XY")        # 从位置 4 写入 → "ABCDXYGH"
                          # 而非预期在文件末尾追加!

# ❌ 更危险的场景
with open("data.txt", "r+") as f:
    f.write("NEW")       # 从位置 0 写入 → "NEWDEFGH"
    content = f.read()   # 从位置 3 读取 → "DEFGH"
    # 写入和读取的位置交错, 极易出错

# ✅ 正确做法：明确 seek 后再写入
with open("data.txt", "r+") as f:
    content = f.read()
    f.seek(0, 2)          # seek 到文件末尾
    f.write("APPENDED")   # 安全追加

# ✅ 更好：使用 a 模式追加
with open("data.txt", "a") as f:
    f.write("APPENDED")   # a 模式保证写入总在末尾
```

**关键洞察**：`r+` 模式下，`read()` 和 `write()` 共享同一个文件位置指针。每次操作后指针都会移动。如果你不显式 `seek()`，写入位置取决于上次操作结束的位置——这几乎是 bug 的温床。

### 陷阱 5：管道重定向导致 UnicodeEncodeError

```python
# print.py
print("中文输出")

# 直接运行: OK (终端支持中文)
$ python print.py
中文输出

# 管道重定向: 崩溃!
$ python print.py > output.txt
# UnicodeEncodeError: 'cp936' codec can't encode character...
# 因为 stdout 被重定向后, 编码变成了 locale 的默认编码
# 而某些字符在该编码中不存在
```

```python
# ❌ 脆弱代码
print("特殊字符：\u2603 \u2764")  # ☃ ❤

# ✅ 解决方案 1: 使用 sys.stdout.reconfigure (3.7+)
import sys
sys.stdout.reconfigure(encoding='utf-8')
print("特殊字符：\u2603 \u2764")  # 现在安全

# ✅ 解决方案 2: 环境变量
# PYTHONIOENCODING=utf-8 python print.py > output.txt

# ✅ 解决方案 3: Python UTF-8 模式 (3.7+)
# python -X utf8 print.py > output.txt

# ✅ 解决方案 4: 直接操作 buffer
import sys
sys.stdout.buffer.write("特殊字符：☃ ❤\n".encode('utf-8'))
```

**深层原因**：当 stdout 连接到终端时，Python 使用终端编码（`sys.stdout.encoding`）。管道重定向后，`sys.stdout` 变成 `None` 编码——Python 回退到 `locale.getpreferredencoding()`。在 Windows 上，这可能是 `cp936` (GBK)，无法编码所有 Unicode 字符。

### 陷阱 6：大文件一次性读取（OOM）

```python
# ❌ 大文件一次性读入内存
with open("huge.log", "r") as f:      # 10 GB 日志文件
    content = f.read()                # MemoryError 或系统 swap 瘫痪

# ❌ readlines() 同样危险
with open("huge.log", "r") as f:
    lines = f.readlines()             # 同样全部加载到内存

# ✅ 逐行迭代（内存恒定）
with open("huge.log", "r") as f:
    for line in f:                    # 每次只加载一行到内存
        if "ERROR" in line:
            process_line(line)

# ✅ 分块读取二进制文件
BLOCK_SIZE = 64 * 1024  # 64 KB
with open("huge.bin", "rb") as f:
    while chunk := f.read(BLOCK_SIZE):   # walrus operator (3.8+)
        process_chunk(chunk)

# ✅ 使用 readline() 手动控制
with open("huge.log", "r") as f:
    while True:
        line = f.readline()
        if not line:
            break
        if "ERROR" in line:
            process_line(line)
```

**为什么 `for line in f` 是安全的**？CPython 的 `TextIOWrapper` 实现了一个**内部行缓冲区**：

```
TextIOWrapper.__next__():
  1. 调用 _ReadAheadBuffer 获取下一行
  2. 内部 BufferedReader 预读 DEFAULT_BUFFER_SIZE(8KB) 字节
  3. 在缓冲区内按 \n 分割, 返回一行
  4. 已消费的行从缓冲区移除 → 内存释放
  5. 缓冲区耗尽时, 重新从底层 FileIO 读取 8KB
```

这意味着无论文件多大，内存中的数据不超过 `DEFAULT_BUFFER_SIZE + 当前行长度`。

```python
bad_data = b"Hello\xFFWorld"

f = io.TextIOWrapper(io.BytesIO(bad_data), encoding='utf-8', errors='replace')

f = io.TextIOWrapper(io.BytesIO(bad_data), encoding='utf-8', errors='backslashreplace')
```

### 题目 3：换行符处理

Python 的 `open()` 有一个 `newline` 参数。请解释 `newline=None`（默认）和 `newline=''` 的区别：

```python
# 文件内容: b"line1\r\nline2\rline3\n"
```

<details>
<summary>参考答案</summary>

```python
import io

content = b"line1\r\nline2\rline3\n"

# newline=None (默认): 通用换行符模式
# 读取时: \r\n, \r, \n 全部转换为 \n
f = io.TextIOWrapper(io.BytesIO(content), encoding='utf-8', newline=None)
lines = f.readlines()
print(lines)  # ['line1\n', 'line2\n', 'line3\n']
# 三种换行符统一为 \n

# newline='': 不做任何转换
# 读取时: 原样保留 \r\n, \r, \n
f = io.TextIOWrapper(io.BytesIO(content), encoding='utf-8', newline='')
lines = f.readlines()
print(lines)  # ['line1\r\n', 'line2\r', 'line3\n']
# 但 readline() 仍能按 \n 分割行!

# newline='\n': 只认 \n 为换行符
# 读取时: \r\n 不会被合并, \r 不会被识别为换行
f = io.TextIOWrapper(io.BytesIO(content), encoding='utf-8', newline='\n')
lines = f.readlines()
print(lines)  # ['line1\r\n', 'line2\rline3\n']
# \r\n 保留原样, \r 被视为普通字符

# newline='\r\n': 写入时只将 \n 转换为 \r\n
# (主要用于 Windows 特定需求)
```

**写入时的行为**：

| newline值 | 写入时 `\n` 的转换 |
|:----------|:-------------------|
| `None` | `\n` → `os.linesep`（Windows: `\r\n`, Unix: `\n`）|
| `''` | 不转换，原样写入 `\n` |
| `'\n'` | 不转换 |
| `'\r\n'` | `\n` → `\r\n` |

</details>

### 题目 4：缓冲区行为

以下代码中，`data1` 和 `data2` 各触发了几次系统调用（`read()` syscall）？

```python
import io
f_raw = io.FileIO("data.bin", "r")     # 假设文件 20KB
f_buf = io.BufferedReader(f_raw, buffer_size=8192)

data1 = f_buf.read(100)   # 第 1 次
data2 = f_buf.read(100)   # 第 2 次
```

如果随后连续调用 81 次 `f_buf.read(100)`，总共触发了多少次系统调用？

<details>
<summary>参考答案</summary>

```python
# 第 1 次 f_buf.read(100):
#   缓冲区空 → 触发 1 次 syscall: read(fd, 8192)
#   读取 8192 字节到缓冲区, 返回前 100 字节
#   系统调用: 1 次

# 第 2 次 f_buf.read(100):
#   缓冲区还有 8092 字节 → 直接从缓冲区取 100 字节
#   系统调用: 0 次

# 连续 81 次 read(100) 的总 syscall 分析:
# 每次读取 100 字节, 缓冲区 8192 字节
# 每 8192/100 ≈ 81.92 次 read(100) 耗尽一个缓冲区
# 即: 每 82 次 read(100) 触发 1 次 syscall (除了第一次)
#
# 总共: 1 (初始) + 82 (次) = 83 次 read(100) 调用
# 需要读取: 83 × 100 = 8300 字节 (超过一个缓冲区)
#
# 第一次 syscall 加载 8192 字节
# 前 81 次 read(100) 消耗 8100 字节, 缓冲区剩 92 字节
# 第 82 次 read(100): 缓冲区只有 92 字节, 不足 100
#   → 返回 92 字节 + 触发第 2 次 syscall 读取 8192 字节
#   → 从新缓冲区取 8 字节, 凑够 100 字节
# 所以第 82 次触发了 1 次附加 syscall

# 总结:
# data1 → 1 次 syscall
# data2 → 0 次 syscall
# 后续 81 次 → 1 次附加 syscall (在第 82 次时)
# 总共: 2 次 syscall (对比无缓冲: 83 次 syscall → 97.6% 减少!)
```

</details>

### 题目 5：struct 二进制解析

给定一个二进制文件头格式（小端序）：
- 4 字节魔数：`b"PK\x03\x04"` (ZIP 文件头)
- 2 字节版本需要: `uint16`
- 2 字节标志位: `uint16`
- 4 字节压缩方法: `uint32`
- 4 字节 CRC-32: `uint32`

请用 `struct` 模块解析这个头部，并计算偏移量。

<details>
<summary>参考答案</summary>

```python
import struct

# 格式字符串：'< 4s H H I I'
# < = little-endian
# 4s = 4 字节 bytes
# H = unsigned short (2 字节)
# I = unsigned int (4 字节)
HEADER_FORMAT = '<4sHHII'
HEADER_SIZE = struct.calcsize(HEADER_FORMAT)
print(f"头部大小: {HEADER_SIZE} 字节")  # 4+2+2+4+4 = 16 字节

# 模拟一个 ZIP 文件头部
magic = b"PK\x03\x04"
version = 20          # 版本 2.0
flags = 0x0800        # 含数据描述符
compression = 8       # deflate
crc32 = 0xDEADBEEF

header_bytes = struct.pack(HEADER_FORMAT, magic, version, flags, compression, crc32)
print(f"头部二进制: {header_bytes.hex()}")
# 504b0304 1400 0008 08000000 efbeadde

# 从文件读取并解析
import io
file_data = header_bytes + b"\x00" * 100  # 模拟完整文件
f = io.BytesIO(file_data)

# 1. 使用 unpack 直接解析
raw_header = f.read(HEADER_SIZE)
magic, version, flags, compression, crc32 = struct.unpack(HEADER_FORMAT, raw_header)
print(f"魔数: {magic!r}")          # b'PK\x03\x04'
print(f"版本: {version / 10:.1f}")  # 2.0
print(f"标志: 0x{flags:04x}")       # 0x0800
print(f"压缩: {compression}")       # 8 (deflate)
print(f"CRC-32: 0x{crc32:08x}")     # 0xdeadbeef

# 2. 使用 unpack_from 避免读取（零拷贝思路）
f.seek(0)
all_data = f.read()
values = struct.unpack_from(HEADER_FORMAT, all_data, offset=0)
# unpack_from 不创建新的 bytes 切片, 直接从 buffer 读取

# 3. 结合 memoryview 实现零拷贝解析
mv = memoryview(all_data)
header_mv = mv[:HEADER_SIZE]            # 不复制数据!
values = struct.unpack_from(HEADER_FORMAT, header_mv)
# 适用于大文件: 不需要为每个字段复制内存

# 字段偏移量计算
offsets = struct.calcsize  # 无法直接获取, 手动计算
field_offsets = {
    'magic':       0,
    'version':     4,
    'flags':       6,
    'compression': 8,
    'crc32':       12,
}
print(f"总偏移: {HEADER_SIZE}")  # 16 — 下一个字段从这里开始
```

</details>
